cmake_minimum_required(VERSION 3.24)
project(VJoltPhysicsConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This example uses the current Windows backend loader (MSVC x86/x64)")
endif()
if(NOT TARGET VJolt::SDK)
  if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../../cmake/VJoltSDK.cmake")
    include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/VJoltSDK.cmake")
  else()
    find_package(VJoltSDK CONFIG REQUIRED)
  endif()
endif()
add_library(vjolt_attractor STATIC attractor.cpp)
target_link_libraries(vjolt_attractor PUBLIC VJolt::SDK)
target_include_directories(vjolt_attractor PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(vjolt_attractor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(vjolt_attractor PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
