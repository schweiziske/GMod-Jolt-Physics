cmake_minimum_required(VERSION 3.24)
project(JoltFeatureExample LANGUAGES CXX)

get_filename_component(example_repo_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(VJOLT_ROOT "${example_repo_root}" CACHE PATH "GMod Jolt Physics source checkout")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The VJolt backend loader currently requires Windows/MSVC")
endif()
include("${VJOLT_ROOT}/cmake/VJoltSDK.cmake")
set(GMOD_COMMON "" CACHE PATH "External garrysmod_common checkout")
if(NOT EXISTS "${GMOD_COMMON}/include/GarrysMod/Lua/Interface.h")
  message(FATAL_ERROR "Set GMOD_COMMON to a matching garrysmod_common checkout")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(gmod_platform_suffix win64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(gmod_platform_suffix win32)
else()
  message(FATAL_ERROR "Unsupported pointer width")
endif()

add_library(joltfeature SHARED source/main.cpp)
target_include_directories(joltfeature PRIVATE "${GMOD_COMMON}/include")
set_target_properties(joltfeature PROPERTIES PREFIX "" OUTPUT_NAME "gmsv_joltfeature_${gmod_platform_suffix}")
target_link_libraries(joltfeature PRIVATE VJolt::SDK)
# Use the example contract before it has been copied into the public SDK.
target_include_directories(joltfeature BEFORE PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../public")
target_compile_definitions(joltfeature PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
install(TARGETS joltfeature LIBRARY DESTINATION lua/bin RUNTIME DESTINATION lua/bin)
