cmake_minimum_required(VERSION 3.24)
project(JoltSoftBody LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/VJoltSDK.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GMOD_COMMON "" CACHE PATH "garrysmod_common checkout")
if(NOT MSVC OR NOT (CMAKE_SIZEOF_VOID_P EQUAL 4 OR CMAKE_SIZEOF_VOID_P EQUAL 8))
  message(FATAL_ERROR "GMod Windows Win32 or x64 is required")
endif()
if(NOT EXISTS "${GMOD_COMMON}/include/GarrysMod/Lua/Interface.h")
  message(FATAL_ERROR "Set GMOD_COMMON")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(gmod_platform_suffix win64)
else()
  set(gmod_platform_suffix win32)
endif()
set(gmod_module gmsv_joltsoftbody_${gmod_platform_suffix})
add_library(${gmod_module} SHARED source/main.cpp source/lua_mesh.cpp source/lua_snapshot.cpp)
target_include_directories(${gmod_module} PRIVATE "${GMOD_COMMON}/include")
target_link_libraries(${gmod_module} PRIVATE VJolt::SDK)
target_compile_definitions(${gmod_module} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(${gmod_module} PRIVATE /W4 /Zi)
target_link_options(${gmod_module} PRIVATE /DEBUG)
set_target_properties(${gmod_module} PROPERTIES PREFIX "")
